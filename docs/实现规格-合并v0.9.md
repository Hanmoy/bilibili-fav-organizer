### 0.5 Worker 源码字符串（T3 的 L2 压测 worker，内联 Blob 构造）

```ts
// classify.ts 顶部（模块级常量）
const WORKER_SRC = `
self.onmessage = function(e) {
  var pattern = e.data.pattern;
  var flags = e.data.flags;
  var sample = e.data.sample;
  try {
    var re = new RegExp(pattern, flags);
    // 对"最坏输入"跑 test：重复字符 + 不匹配尾，触发灾难性回溯
    re.test(sample);
    self.postMessage({ safe: true });
  } catch (err) {
    self.postMessage({ safe: false, error: String(err) });
  }
};
`;

/** L2: Blob Worker 压测正则（CSP 失败时降级返回 'unknown'） */
function stressTestRegex(pattern: string, flags: string, timeoutMs = 300): Promise<'safe' | 'unsafe' | 'unknown'> {
  return new Promise((resolve) => {
    let worker: Worker | null = null;
    let settled = false;
    const done = (r: 'safe' | 'unsafe' | 'unknown') => {
      if (settled) return;
      settled = true;
      if (worker) worker.terminate();
      resolve(r);
    };
    try {
      const blob = new Blob([WORKER_SRC], { type: 'application/javascript' });
      worker = new Worker(URL.createObjectURL(blob));
      // 构造最坏输入：pattern 首字符重复 30 次 + 一个不匹配尾字符
      const firstChar = (pattern.match(/[a-zA-Z0-9]/) || ['a'])[0];
      const sample = firstChar.repeat(30) + '!';
      worker.onmessage = (e: MessageEvent) => done(e.data.safe ? 'safe' : 'unsafe');
      worker.onerror = () => done('unknown');
      worker.postMessage({ pattern, flags, sample });
      setTimeout(() => done('unsafe'), timeoutMs);
    } catch {
      // CSP 拦截 blob worker 或环境不支持
      done('unknown');
    }
  });
}
```

```ts
/** 归一化单条规则：补默认字段、类型校验、空 pattern 强制禁用 */
function normalizeRule(r: any, fallbackId: string): ClassifyRule {
  const pattern = typeof r?.pattern === 'string' ? r.pattern : '';
  const scope: RuleScope = r?.scope === 'title_intro' || r?.scope === 'upper' ? r.scope : 'title';
  return {
    id: typeof r?.id === 'string' && r.id ? r.id : fallbackId,

function normalizeRules(rules: any[]): ClassifyRule[] {
  if (!Array.isArray(rules)) return [];
  return rules.map((r, i) => normalizeRule(r, `migrated_${Date.now().toString(36)}_${i}`));
```

#### 3.2 新增 matchOne（纯函数，classifyByRules 与 testRules 共用内核）
```ts
/** 按 scope 取待匹配文本 */
function ruleText(m: FavMedia, scope: RuleScope): string {
  if (scope === 'upper') return m.upper?.name || '';
  if (scope === 'title_intro') return `${m.title} ${m.intro || ''}`;
  return m.title;
}

/** 单条规则对单条视频的匹配判定（纯函数） */
export function matchOne(r: ClassifyRule, m: FavMedia): boolean {
  if (!r.enabled || !r.pattern) return false; // 禁用/空 pattern 跳过
  try {
    if (r.isRegex) {
      return new RegExp(r.pattern, ruleFlags(r)).test(ruleText(m, r.scope || 'title'));
    }
    const text = ruleText(m, r.scope || 'title');
    const pat = r.pattern;
    return r.caseSensitive ? text.includes(pat) : text.toLowerCase().includes(pat.toLowerCase());
  } catch {
    return false; // 非法正则视为不匹配（invalid 由 validateRules 收集）
  }
}
```

#### 3.3 classifyByRules 改用 matchOne + 缓存 + 收集 invalid
**改动位置**：classify.ts:17-39 整体重写。
```ts
// 模块级 RegExp 缓存（key = pattern|flags）
const regexCache = new Map<string, RegExp>();

/** 规则引擎：命中即归类，返回 { 结果, 剩余, invalidIds } */
export function classifyByRules(
  medias: FavMedia[],
  rules: ClassifyRule[],
): { result: CategoryMap; rest: FavMedia[]; invalidIds: string[] } {
  const result: CategoryMap = {};
  const rest: FavMedia[] = [];
  const invalidIds: string[] = [];
  // 预编译正则，收集 invalid
  const compiled: Array<{ r: ClassifyRule; re: RegExp | null }> = rules.map((r) => {
    if (!r.enabled || !r.pattern) return { r, re: null };
    if (!r.isRegex) return { r, re: null };
    const key = `${r.pattern}|${ruleFlags(r)}`;
    if (regexCache.has(key)) return { r, re: regexCache.get(key)! };
    try {
      const re = new RegExp(r.pattern, ruleFlags(r));
      regexCache.set(key, re);
      return { r, re };
    } catch {
      invalidIds.push(r.id);
      return { r, re: null };
    }
  });
  for (const m of medias) {
    let hit = '';
    for (const { r, re } of compiled) {
      if (!r.enabled || !r.pattern) continue;
      let matched = false;
      if (r.isRegex && re) {
        matched = re.test(ruleText(m, r.scope || 'title'));
      } else if (!r.isRegex) {
        matched = matchOne(r, m);
      }
      if (matched) { hit = r.category; break; }
    }
    if (hit) (result[hit] ||= []).push({ ...m, category: hit });
    else rest.push(m);
  }
  return { result, rest, invalidIds };
}
```

#### 3.4 新增 lintRegex（L1 静态扫描）
```ts
/** L1: 静态扫描正则源码，拦截已知灾难性回溯模式 */
export function lintRegex(pattern: string): { ok: boolean; reason?: string } {
  if (!pattern) return { ok: true };
  // 嵌套量词：(a+)+ / (a*)* / (a+)* 等
  if (/\([^)]*[+*][^)]*\)[+*{]/.test(pattern)) {
    return { ok: false, reason: '检测到嵌套量词（如 (a+)+），极易引发灾难性回溯，请简化' };
  }
  // 超长重复：a{100,}（量级过大）
  if (/\{\d{3,},\}/.test(pattern)) {
    return { ok: false, reason: '重复次数过大（{100+}），可能性能不佳，请缩小范围' };
  }
  return { ok: true };
}
```

#### 3.5 新增 validateRules（校验 + 调 L1+L2）
```ts
export interface RuleValidation {
  valid: boolean;
  invalid: Array<{ id: string; reason: string }>;
}

/**
 * 校验规则集：空 pattern 标记、非法正则标记、ReDoS 风险标记
 * @param rules 待校验规则
 * @param stressTest 是否对正则跑 L2 Worker 压测（保存时 true，运行时 false）
 */
export async function validateRules(
  rules: ClassifyRule[],
  stressTest = false,
): Promise<RuleValidation> {
  const invalid: Array<{ id: string; reason: string }> = [];
  for (const r of rules) {
    if (!r.pattern) {
      invalid.push({ id: r.id, reason: '匹配内容为空（已自动禁用）' });
      continue;
    }
    if (r.isRegex) {
      // 语法校验
      try {
        new RegExp(r.pattern, ruleFlags(r));
      } catch (e) {
        invalid.push({ id: r.id, reason: `正则语法错误：${e instanceof Error ? e.message : e}` });
        continue;
      }
      // L1 静态扫描
      const lint = lintRegex(r.pattern);
      if (!lint.ok) {
        invalid.push({ id: r.id, reason: lint.reason! });
        continue;
      }
      // L2 Worker 压测（仅保存时）
      if (stressTest) {
        const stress = await stressTestRegex(r.pattern, ruleFlags(r));
        if (stress === 'unsafe') {
          invalid.push({ id: r.id, reason: '正则存在回溯炸弹风险（压测超时），请简化或改用关键词匹配' });
        }
        // 'unknown'(CSP 降级) 不阻断，仅 L1 已过
      }
    }
  }
  return { valid: invalid.length === 0, invalid };
}
```

```ts
  function bindRuleEvents() {
    // 编辑/展开
    ruleList.querySelectorAll('[data-edit]').forEach((b) => {
      b.addEventListener('click', () => {
        const id = (b as HTMLElement).dataset.edit!;
        const r = state.config.rules.find((x) => x.id === id);
        if (!r) return;
        editingId = id;
        editingDraft = { ...r };
        renderRules();
      });
    });
    // 删除（改用 id 定位）
    ruleList.querySelectorAll('[data-del]').forEach((b) => {
      b.addEventListener('click', () => {
        const id = (b as HTMLElement).dataset.del!;
        const idx = state.config.rules.findIndex((x) => x.id === id);
        if (idx < 0) return;
        state.config.rules.splice(idx, 1);
        persist();
        renderRules();
      });
    });
    // 编辑态字段同步
    ruleList.querySelectorAll('[data-field]').forEach((el) => {
      const field = (el as HTMLElement).dataset.field!;
      const handler = () => {
        const input = el as HTMLInputElement | HTMLSelectElement;
        if (field === 'isRegex') {
          editingDraft.isRegex = (input as HTMLInputElement).value === 'true';
        } else if (field === 'category') {
          const val = (input as HTMLSelectElement).value;
          editingDraft.category = val === '__custom__' ? '' : val;
          const customInput = ruleList.querySelector('[data-field="categoryCustom"]') as HTMLElement;
          if (customInput) customInput.style.display = val === '__custom__' ? '' : 'none';
        } else if (field === 'categoryCustom') {
          editingDraft.category = (input as HTMLInputElement).value;
        } else if (typeof editingDraft[field as keyof ClassifyRule] === 'boolean' || field === 'caseSensitive' || field === 'enabled') {
          (editingDraft as any)[field] = (input as HTMLInputElement).checked;
        } else {
          (editingDraft as any)[field] = (input as HTMLInputElement).value;
        }
      };
      el.addEventListener('input', handler);
      el.addEventListener('change', handler);
    });
    // 保存
    ruleList.querySelectorAll('[data-action="save"]').forEach((b) => {
      b.addEventListener('click', async () => {
        const id = editingId!;
        const r = state.config.rules.find((x) => x.id === id);
        if (!r) return;
        const draft = { ...r, ...editingDraft } as ClassifyRule;
        // 合并 categoryCustom
        if (!draft.category) {
          const custom = (ruleList.querySelector('[data-field="categoryCustom"]') as HTMLInputElement)?.value || '';
          draft.category = custom;
        }
        // 校验
        const { valid, invalid } = await validateRules([draft], true); // stressTest=true
        if (!valid) {
          const reason = invalid[0]?.reason || '校验失败';
          (ruleList.querySelector('#wb-rule-test-result') as HTMLElement).innerHTML = `<span style="color:var(--wb-red)">${esc(reason)}</span>`;
          return;
        }
        Object.assign(r, draft);
        persist();
        editingId = null;
        editingDraft = {};
        renderRules();
        toast('规则已保存');
      });
    });
    // 取消
    ruleList.querySelectorAll('[data-action="cancel"]').forEach((b) => {
      b.addEventListener('click', () => { editingId = null; editingDraft = {}; renderRules(); });
    });
    // 上移/下移
    ruleList.querySelectorAll('[data-action="up"],[data-action="down"]').forEach((b) => {
      b.addEventListener('click', () => {
        const id = editingId!;
        const idx = state.config.rules.findIndex((x) => x.id === id);
        if (idx < 0) return;
        const dir = (b as HTMLElement).dataset.action === 'up' ? -1 : 1;
        const ni = idx + dir;
        if (ni < 0 || ni >= state.config.rules.length) return;
        [state.config.rules[idx], state.config.rules[ni]] = [state.config.rules[ni], state.config.rules[idx]];
        persist();
        renderRules();
      });
    });
  }
```

### 3.3 关键风险提示

1. **Blob Worker 真机验证**：T3 的 stressTestRegex 需在 B站登录态(www.bilibili.com)真机验证 `new Worker(blobUrl)` 是否被 CSP 拦截。若被拦，确认 catch 降级返回 'unknown' 不崩（已设计降级，但务必真机测一次）。
2. **classifyByRules 返回值变更**：T3 给 classifyByRules 增了 `invalidIds` 返回，classifyAll 两处调用(classify.ts:93、323)必须同步解构，否则编译报错。
3. **onTaxonomyConfirm 的 Promise 挂起**：A3 用 `window.addEventListener('wb:taxonomy')` 等待 UI 事件 resolve Promise。若用户在确认态刷新页面，Promise 永不 resolve——但刷新会重置 state.task，classifyAll 调用链随之废弃，无泄漏。确认态由 task.taxonomyConfirm 持久化，刷新后 renderOrganize 重新渲染确认 UI。
4. **clean 接口不可逆**：E2 的"一键清空"是删除操作，confirm 二次确认必须醒目（已加 ⚠️ + "不可逆"措辞）。
